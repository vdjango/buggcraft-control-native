cmake_minimum_required(VERSION 3.25)
project(buggcraft_control VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(CONTROL_LTO "Release link time optimization" ON)
option(CONTROL_SANITIZERS "Address/undefined behavior sanitizers" OFF)
find_package(Boost 1.74 REQUIRED)
find_package(OpenSSL 3.0 REQUIRED)
find_package(PostgreSQL 14 REQUIRED)
find_package(simdjson CONFIG REQUIRED)
find_package(nlohmann_json 3.10 CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(CURL 7.85 REQUIRED)
add_library(control-domain src/json.cpp src/crypto.cpp src/config.cpp src/database.cpp
  src/service.cpp src/enrollment.cpp src/placement.cpp src/social.cpp src/probe.cpp src/browser.cpp src/browser_transport.cpp src/social_import.cpp src/activity.cpp)
target_include_directories(control-domain PUBLIC include)
target_link_libraries(control-domain PUBLIC OpenSSL::Crypto PostgreSQL::PostgreSQL
  simdjson::simdjson nlohmann_json::nlohmann_json Threads::Threads Boost::headers OpenSSL::SSL CURL::libcurl)
if(WIN32)
  target_compile_definitions(control-domain PUBLIC _WIN32_WINNT=0x0602 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(control-domain PUBLIC ws2_32 mswsock)
endif()
target_compile_definitions(control-domain PUBLIC CONTROL_VERSION="${PROJECT_VERSION}")
add_executable(buggcraft-control src/main.cpp src/server.cpp)
if(UNIX AND NOT APPLE)
  set_target_properties(buggcraft-control PROPERTIES
    BUILD_RPATH "$ORIGIN/../lib" INSTALL_RPATH "$ORIGIN/../lib")
endif()
target_link_libraries(buggcraft-control PRIVATE control-domain Boost::headers OpenSSL::SSL)
if(WIN32)
  target_compile_definitions(buggcraft-control PRIVATE _WIN32_WINNT=0x0602 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(buggcraft-control PRIVATE ws2_32 mswsock)
endif()
include(CTest)
add_executable(control-units tests/units.cpp)
target_link_libraries(control-units PRIVATE control-domain)
add_test(NAME control-units COMMAND control-units)
add_executable(control-benchmark benchmarks/json.cpp)
target_link_libraries(control-benchmark PRIVATE control-domain)
foreach(target control-domain buggcraft-control control-units control-benchmark)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    if(CONTROL_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
if(CONTROL_LTO AND NOT CONTROL_SANITIZERS)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_ok)
  if(ipo_ok)
    set_property(TARGET control-domain buggcraft-control control-units control-benchmark
      PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  endif()
endif()
install(TARGETS buggcraft-control RUNTIME DESTINATION bin)
install(FILES sql/001-initial.sql DESTINATION share/buggcraft-control)
